cmake_minimum_required(VERSION 3.22.1)
project(bkawrapper LANGUAGES C CXX)

# Define source files
add_library(bkawrapper SHARED
    ultra/NativeBridge.cpp
    ultra/otr_builder.cpp
    ultra/audio_helper.cpp
    ultra/exceptasm.cpp
    emulator/resource_mgr.cpp
    emulator/pi_hle.cpp
    emulator/stubs.cpp
    tools/rare_decompression.cpp
)

# Configure Include Paths
target_include_directories(bkawrapper PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/include/2.0L
    ${CMAKE_CURRENT_SOURCE_DIR}/include/2.0L/PR
    ${CMAKE_CURRENT_SOURCE_DIR}/tools
    ${CMAKE_CURRENT_SOURCE_DIR}/ultra
    ${CMAKE_CURRENT_SOURCE_DIR}/emulator
    ${CMAKE_CURRENT_SOURCE_DIR}/game_src
)

# Required for N64 headers to expose correct logic
target_compile_definitions(bkawrapper PRIVATE _LANGUAGE_C)

# --- FIX START: Generate a single Master Header ---
# We create a temporary header file that includes all necessary headers.
# This avoids the "multiple output files" error by passing only ONE -include flag to the compiler.

set(GLOBAL_PCH "${CMAKE_CURRENT_BINARY_DIR}/global_pch.h")
file(WRITE ${GLOBAL_PCH} "/* Auto-generated by CMake to fix force-include issues */
#pragma once
#include <stddef.h>
#include <stdint.h>
#include <string.h>
/* Include the N64 types using the absolute path to ensure it is found */
#include \"${CMAKE_CURRENT_SOURCE_DIR}/include/2.0L/PR/ultratypes.h\"
")

# Force include the single master header we just generated
target_compile_options(bkawrapper PRIVATE "-include" "${GLOBAL_PCH}")
# --- FIX END ---

# Linker setup
find_library(log-lib log)
find_library(android-lib android)
find_library(jnigraphics-lib jnigraphics)
find_library(z-lib z)

target_link_libraries(bkawrapper
    ${log-lib}
    ${android-lib}
    ${jnigraphics-lib}
    ${z-lib}
    atomic
    m
)
