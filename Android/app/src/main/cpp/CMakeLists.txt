cmake_minimum_required(VERSION 3.22.1)
project(bkawrapper LANGUAGES C CXX)

# 1. Dynamic Path Calculation
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(DECOMP_ROOT "${REPO_ROOT}/decomp-files")

message(STATUS "[BK] REPO_ROOT: ${REPO_ROOT}")
message(STATUS "[BK] DECOMP_ROOT: ${DECOMP_ROOT}")

# 2. Define Flags as a LIST (No quotes around the space-separated values)
set(COMMON_FLAGS 
    -w 
    -fcommon 
    -O2 
    -D_FINALROM 
    -D_LANGUAGE_C 
    -D_ULTRATYPES_H_ 
    -DANDROID
)

# 3. Header Bridge
set(BRIDGE_HEADER "${DECOMP_ROOT}/include/n64_types.h")

# 4. Include Search Paths
include_directories(SYSTEM
    ${DECOMP_ROOT}/include
    ${DECOMP_ROOT}/include/2.0L
    ${DECOMP_ROOT}/include/2.0L/PR
    ${DECOMP_ROOT}/src/core1
)

# 5. Source Discovery
file(GLOB_RECURSE CORE1_SOURCES "${DECOMP_ROOT}/src/core1/*.c")
list(LENGTH CORE1_SOURCES SRC_COUNT)
message(STATUS "[BK] Found ${SRC_COUNT} core source files")

# Filter logic
list(FILTER CORE1_SOURCES EXCLUDE REGEX ".*memory.c")
list(FILTER CORE1_SOURCES EXCLUDE REGEX ".*n_reverb.*")

# 6. Build the Library
# We pass NativeBridge.cpp directly, which is now generated by the Python script
add_library(bkawrapper SHARED 
    ${CORE1_SOURCES} 
    NativeBridge.cpp
)

# 7. Apply Options (CMake will now correctly expand these as separate arguments)
target_compile_options(bkawrapper PRIVATE ${COMMON_FLAGS} -include "${BRIDGE_HEADER}")

target_link_libraries(bkawrapper log android z GLESv2 EGL m atomic)
