cmake_minimum_required(VERSION 3.22.1)
project("bkawrapper")

# 1. Set Compiler Standards
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)

# 2. Locate Assets and Headers
include_directories(include)
include_directories(src)

# 3. Recursive Discovery of Harmonized Source Files
# This finds all game logic files moved by the Harmonizer script
file(GLOB_RECURSE GAME_SOURCES 
    "src/*.c"
    "src/*.cpp"
)

# 4. Define the Native Library
add_library(bkawrapper SHARED
    # Core Android Bridge
    ultra/NativeBridge.cpp
    
    # All game engine files discovered by glob
    ${GAME_SOURCES}
)

# 5. Link Libraries
find_library(log-lib log)
find_package(ZLIB REQUIRED)

target_link_libraries(bkawrapper 
    ${log-lib} 
    ${ZLIB_LIBRARIES}
    android
)

# 6. Apply the Symbol Map generated by the Harmonizer
target_link_options(bkawrapper PRIVATE "-Wl,--version-script,${CMAKE_CURRENT_SOURCE_DIR}/symbol_map.txt")

# 7. Add N64 Math Definitions
target_compile_definitions(bkawrapper PRIVATE 
    _LANGUAGE_C 
    NON_MATCHING 
    F3DEX_GBI_2
)
