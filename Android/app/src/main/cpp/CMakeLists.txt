cmake_minimum_required(VERSION 3.22.1)
project(bkawrapper LANGUAGES C CXX)

# 1. Gather all Game Source Files (The missing pieces)
# This finds all .c and .cpp files in the src/ directory recursively
file(GLOB_RECURSE GAME_SOURCES 
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"
)

# 2. Define the library including both wrapper and game sources
add_library(bkawrapper SHARED
    ${GAME_SOURCES}
    ultra/NativeBridge.cpp
    ultra/otr_builder.cpp
    ultra/audio_helper.cpp
    ultra/exceptasm.cpp
    emulator/resource_mgr.cpp
    emulator/pi_hle.cpp
    emulator/stubs.cpp
    tools/rare_decompression.cpp
)

# 3. Configure Include Paths
target_include_directories(bkawrapper PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/include/2.0L
    ${CMAKE_CURRENT_SOURCE_DIR}/include/2.0L/PR
    ${CMAKE_CURRENT_SOURCE_DIR}/tools
    ${CMAKE_CURRENT_SOURCE_DIR}/ultra
    ${CMAKE_CURRENT_SOURCE_DIR}/emulator
    ${CMAKE_CURRENT_SOURCE_DIR}/src
)

# Required for N64 headers
target_compile_definitions(bkawrapper PRIVATE _LANGUAGE_C)

# --- Force-Include Master Header ---
set(GLOBAL_PCH "${CMAKE_CURRENT_BINARY_DIR}/global_pch.h")
file(WRITE ${GLOBAL_PCH} "/* Auto-generated by CMake */
#pragma once
#include <stddef.h>
#include <stdint.h>
#include <string.h>
#include \"${CMAKE_CURRENT_SOURCE_DIR}/include/2.0L/PR/ultratypes.h\"
")

target_compile_options(bkawrapper PRIVATE "-include" "${GLOBAL_PCH}")

# 4. Linker setup
find_library(log-lib log)
find_library(android-lib android)
find_library(jnigraphics-lib jnigraphics)
find_library(z-lib z)

target_link_libraries(bkawrapper
    ${log-lib}
    ${android-lib}
    ${jnigraphics-lib}
    ${z-lib}
    atomic
    m
)
